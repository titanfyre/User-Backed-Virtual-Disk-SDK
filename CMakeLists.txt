cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(UserBackedDisk VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio C++ on Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument user mode" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(UserDisk SDK/Disk.cpp)
target_link_libraries(UserDisk PUBLIC setupapi cfgmgr32 winhttp crypt32 advapi32)
add_executable(UserBackedDisk WIN32 App/Main.cpp)
target_link_libraries(UserBackedDisk PRIVATE UserDisk user32 gdi32 comctl32 comdlg32 shell32)
target_link_options(UserBackedDisk PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(UserBackedDiskCLI App/Console.cpp)
target_link_libraries(UserBackedDiskCLI PRIVATE UserDisk)
add_executable(UserBackedDiskExample Examples/memory.cpp)
target_link_libraries(UserBackedDiskExample PRIVATE UserDisk)
add_library(DiskSetup InstallerUI/Setup.cpp)
target_link_libraries(DiskSetup PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft signed driver package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanUserBackedDisk.inf TitanUserBackedDisk.cat TitanUserBackedDisk.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete signed package")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanUserBackedDisk.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanUserBackedDisk.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanUserBackedDisk.sys\"\n")
endif()
add_executable(UserBackedDiskSetup WIN32 InstallerUI/Main.cpp ${PACKAGE_RESOURCE})
add_executable(UserBackedDiskSetupConsole InstallerConsole/Main.cpp ${PACKAGE_RESOURCE})
foreach(target UserBackedDiskSetup UserBackedDiskSetupConsole)
 target_link_libraries(${target} PRIVATE DiskSetup user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
add_executable(UserBackedDiskTests Tests/Protocol.cpp)
target_link_libraries(UserBackedDiskTests PRIVATE UserDisk)
add_executable(UserBackedDiskInstallerTests Tests/Installer.cpp)
target_link_libraries(UserBackedDiskInstallerTests PRIVATE DiskSetup)
enable_testing()
add_test(NAME DiskProtocolAndBackends COMMAND UserBackedDiskTests)
add_test(NAME InstallerIdentityAndTrustRefusal COMMAND UserBackedDiskInstallerTests)
foreach(target UserBackedDisk UserBackedDiskCLI UserBackedDiskExample UserBackedDiskTests UserBackedDiskInstallerTests UserBackedDiskSetup UserBackedDiskSetupConsole)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
